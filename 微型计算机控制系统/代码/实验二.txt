    Case 5
        x0 = CenterX + QiDianX * Stepx
        y0 = CenterY - QiDianY * Stepy
        
    Case 6
        x0 = CenterX - QiDianX * Stepx
        y0 = CenterY - QiDianY * Stepy
        
   Case 7
        x0 = CenterX - QiDianX * Stepx
        y0 = CenterY + QiDianY * Stepy
        
    Case 8
        x0 = CenterX + QiDianX * Stepx
        y0 = CenterY + QiDianY * Stepy
        
    Case Else
    End Select
    Timer1.Enabled = True

End Sub
