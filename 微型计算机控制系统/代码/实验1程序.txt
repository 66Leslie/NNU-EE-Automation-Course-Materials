Private Sub Command1_Click()
    
    
    Dim Str As String
    ZhongDianX = Val(ZhongdianXTxT.Text)

        xoy = 1

        xoy = 4

        xoy = 3

    ZhongDianY = Abs(ZhongDianY)

    CenterY = Picture1.ScaleHeight \ 2
    
    Picture1.Cls
    Picture1.BackColor = QBColor(15)
    Picture1.ForeColor = QBColor(0)
    Picture1.Line (CenterX, 500)-(CenterX, Picture1.ScaleHeight - 500)
    Picture1.Line (CenterX - 100, 700)-(CenterX, 500)
    Picture1.Line (CenterX + 100, 700)-(CenterX, 500)
    Picture1.Line (500, CenterY)-(Picture1.ScaleWidth - 500, CenterY)
    Picture1.Line (Picture1.ScaleWidth - 700, CenterY + 100)-(Picture1.ScaleWidth - 500, CenterY)
    Picture1.Line (Picture1.ScaleWidth - 700, CenterY - 100)-(Picture1.ScaleWidth - 500, CenterY)
    Picture1.FontSize = 14
    Str = "x"
    Picture1.CurrentX = Picture1.ScaleWidth - 900
    Picture1.CurrentY = CenterY
   
    Picture1.Print Str
    Str = "y"
    Picture1.CurrentX = CenterX - 150
    Picture1.CurrentY = 700
    Picture1.Print Str
    Picture1.ForeColor = QBColor(13)
    
    Nxy = Int(Abs(ZhongDianX) + Abs(ZhongDianY))
    Stepx = (Picture1.ScaleWidth \ 2 - 900) \ (ZhongDianX)
    Stepy = (Picture1.ScaleHeight \ 2 - 900) \ (ZhongDianY)
    If Stepx >= Stepy Then
       Stepx = Stepy
    Else
       Stepy = Stepx
    End If
    x0 = CenterX
    y0 = CenterY
    error1 = 0
    Timer1.Enabled = True
    
End Sub
